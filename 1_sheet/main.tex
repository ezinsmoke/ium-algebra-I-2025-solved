% --- Problem 1 ---
\begin{task}
    \begin{enumerate}
    	\item Докажите, что множества $\mathbb{Z}$ и $2\mathbb{Z}$ изоморфны как абелевы группы относительно сложения. Изоморфны ли они как кольца?
	\item Изоморфны ли как абелевы группы $(\mathbb{R}, +)$ и $(\mathbb{Q}, +)$? $(\mathbb{Q}, +)$ и $(\mathbb{Z}, +)$?
    \end{enumerate}
\end{task}

\sol
\soli{1} Рассмотрим отображение
\[
\varphi: \mathbb{Z} \to 2\mathbb{Z}, \qquad \varphi(n) = 2n.
\]

Оно является:
\begin{enumerate}
    \item \textbf{Гомоморфизмом:} 
    \[
    \varphi(a+b)=2(a+b)=2a+2b=\varphi(a)+\varphi(b).
    \]
    \item \textbf{Инъекцией:} 
    \[
    \varphi(a)=\varphi(b) \implies 2a=2b \implies a=b.
    \]
    \item \textbf{Сюръекцией:} любой элемент $2k \in 2\mathbb{Z}$ имеет прообраз $k \in \mathbb{Z}$, так как $\varphi(k)=2k$.
\end{enumerate}

Следовательно, $\varphi$ — изоморфизм групп. Значит,
\[
\mathbb{Z} \cong 2\mathbb{Z}.
\]

\hfill $\square$

$\mathbb{Z}$ и 2$\mathbb{Z}$ неизоморфны как кольца. Предположим противное: пусть существует изоморфизм колец
\[
\varphi: \mathbb{Z} \to 2\mathbb{Z}.
\]

В любом кольце с единицей изоморфизм переводит единицу в единицу. Кольцо $\mathbb{Z}$ имеет единицу $1$. Значит,
\[
\varphi(1) = 1_{2\mathbb{Z}},
\]
где $1_{2\mathbb{Z}}$ — единичный элемент кольца $2\mathbb{Z}$.

Но в кольце $2\mathbb{Z}$ \emph{нет} единичного элемента: для любого $2k \in 2\mathbb{Z}$ выполнено
\[
(2k) \cdot 2 = 4k \neq 2k
\]
при $k \neq 0$, а при $k=0$ получаем $0\cdot 2 = 0$, но это не единица для других элементов.

Следовательно, $\varphi(1)$ не может существовать в $2\mathbb{Z}$. Противоречие.

Значит, изоморфизма колец не существует:
\[
\mathbb{Z} \not\cong 2\mathbb{Z}.
\]

\hfill $\square$

\newpage

\soli{2} 
\begin{enumerate}
    \item Используем \textbf{кардинальность}:
    \[
    |\mathbb{Q}| = \aleph_0, \qquad |\mathbb{R}| = 2^{\aleph_0}.
    \]
    Любой изоморфизм групп является биекцией, а биекция сохраняет мощность. Так как мощности различны, изоморфизма не существует:
    \[
    (\mathbb{Q}, +) \not\cong (\mathbb{R}, +).
    \]

    \hfill $\square$

    \item В группе $(\mathbb{Z}, +)$ все ненулевые элементы имеют бесконечный порядок, но группа порождается одним элементом ($\mathbb{Z} = \langle 1 \rangle$), то есть является циклической.

    Группа $(\mathbb{Q}, +)$ \emph{не является циклической}: предположим, что $\mathbb{Q} = \langle a/b \rangle$. Тогда все элементы имели бы вид $n \cdot (a/b)$, то есть были бы целыми кратными числа $a/b$. Но, например, число $a/(2b)$ не представимо в таком виде. Противоречие.

    Следовательно, они не изоморфны. \hfill $\square$
\end{enumerate}



% --- Problem 2 ---
\begin{task}
    \begin{enumerate}
	\item Докажите, что любая подгруппа $(\mathbb{Z}, +)$ имеет вид $n \mathbb{Z}$, где $n \geq 0$
	\item Найдите обратный элемент к $[157]$ в кольце $\mathbb{Z}/235\mathbb{Z}$ и все целые решения уравнения $157x + 235y = 11$. 
    \end{enumerate}
\end{task}


\sol
\soli{1}
Пусть $H \leqslant \mathbb{Z}$.

\begin{enumerate}
    \item Если $H = \{0\}$, то $H = 0\mathbb{Z}$.
    
    \item Пусть $H \neq \{0\}$. Тогда в $H$ есть ненулевые элементы. Так как $H$ — подгруппа, вместе с каждым элементом она содержит и его противоположный. Следовательно, в $H$ есть положительные элементы.
    
    Обозначим через $n$ \emph{наименьший положительный} элемент из $H$:
    \[
    n = \min\{ h \in H \mid h > 0 \}.
    \]
    
    Докажем, что $H = n\mathbb{Z}$.
    
    \begin{enumerate}
	\item \textbf{$n\mathbb{Z} \subseteq H$:} Так как $n \in H$, то $n + n + \dots + n \in H$ и $(-n) + (-n) + \dots \in H$. Значит, все кратные $n$ принадлежат $H$.
    
	\item \textbf{$H \subseteq n\mathbb{Z}$:} Возьмём произвольный $h \in H$. Разделим $h$ на $n$ с остатком:
	\[
	h = qn + r, \qquad 0 \leqslant r < n, \quad q \in \mathbb{Z}.
	\]
	Тогда
	\[
	r = h - qn.
	\]
	Здесь $h \in H$ и $qn \in H$ (так как $n \in H$), следовательно, $r \in H$.
    
	Но $0 \leqslant r < n$, а $n$ — наименьший положительный элемент в $H$. Значит, $r$ не может быть положительным. Следовательно, $r = 0$.
    
	Таким образом, $h = qn$, то есть $h \in n\mathbb{Z}$.
    \end{enumerate} 
    Получили $H = n\mathbb{Z}$, где $n \geqslant 0$. 
\end{enumerate}




Итак, любая подгруппа $\mathbb{Z}$ является циклической и порождается своим наименьшим положительным элементом. \hfill $\square$

\soli{2}
\[
\begin{aligned}
    235 &= 1 \cdot 157 + 78 \\
    157 &= 2 \cdot 78 + 1
\end{aligned}
\]

\[
\begin{aligned}
    1 &= 157 - 2 \cdot 78 \\
      &= 157 - 2 \cdot (235 - 157) \\
      &= 3 \cdot 157 - 2 \cdot 235
\end{aligned}
\]

Следовательно, \(157 \cdot 3 + 235 \cdot (-2) = 1\), значит \(n = 3\), \(k = -2\).

\[
    157 \cdot 3 \equiv 1 \pmod{235}
\]

\[
    \boxed{[157]^{-1} = [3]}
\]

\hfill $\square$

Используя вышеописанный алгоритм Евклида находим, что 

\[
\begin{aligned}
    157 \cdot 3 + 235 \cdot (-2) = 1 \, | \, \cdot 11\\
    157 \cdot 33 + 235 \cdot (-22) = 11
\end{aligned}
\]

значит, при $k \in \mathbb{N}$:
\[
    157 \cdot (k + 33) + 235 \cdot (k - 2) = 11
\]

тем самым $x = k + 33, y = k - 2$, при $k \in \mathbb{N}$

\hfill $\square$

% --- Problem 3 ---
\begin{task}
    Убедитесь, что $\mathbb{Z} [ \sqrt{5} ] = \{ a + b \sqrt{5} \, | \, a,b \in \mathbb{Z} \}$   является коммутативным кольцом с единицей без делителей нуля (областью целостности). Покажите, что это кольцо не является факториальным.
\end{task}

\sol
\soli{1}
Множество  
\[
   \{ \mathbb{Z} [ \sqrt{5} ] = \{ a + b \sqrt{5} \, | \,  a,b \in \mathbb{Z} \}
\]
замкнуто относительно сложения, вычитания и умножения:
\[
(a+b\sqrt5)(c+d\sqrt5)=(ac+5bd)+(ad+bc)\sqrt5,
\]
содержит $1=1+0\sqrt5$ и является подкольцом поля $\mathbb R$. Значит, это коммутативное кольцо с единицей и без делителей нуля, то есть область целостности. $\hfill \square$

\soli{2}
Рассмотрим норму
\[
N(a+b\sqrt5)=a^2-5b^2.
\]
Она мультипликативна: $N(\alpha\beta)=N(\alpha)N(\beta)$. Элемент является обратимым тогда и только тогда, когда его норма равна $\pm1$.

В кольце нет элементов нормы $\pm2$, так как сравнение
\[
a^2-5b^2\equiv a^2\pmod 5
\]
не может давать $2$ или $3$ по модулю $5$, а квадраты по модулю $5$ равны $0,1,4$.

Следовательно, любой элемент нормы $\pm4$ неприводим: если $\alpha=\beta\gamma$, то
\[
N(\beta)N(\gamma)=\pm4,
\]
но норм $\pm2$ нет, поэтому один из сомножителей имеет норму $\pm1$, то есть является единицей.

Имеем:
\[
N(2)=4,\qquad N(1+\sqrt5)=-4,\qquad N(\sqrt5-1)=-4,
\]
значит, $2$, $1+\sqrt5$, $\sqrt5-1$ неприводимы.

Но
\[
4=2\cdot 2=(1+\sqrt5)(\sqrt5-1).
\]
Проверим, что $2$ не ассоциирован с $1+\sqrt5$: если бы
\[
2=(c+d\sqrt5)(1+\sqrt5),
\]
то
\[
c+5d=2,\qquad c+d=0,
\]
откуда $d=\frac12\notin\mathbb Z$, противоречие. Аналогично $2$ не ассоциирован с $\sqrt5-1$.

Таким образом, число $4$ имеет два различных разложения на неприводимые множители:
\[
4=2\cdot 2=(1+\sqrt5)(\sqrt5-1).
\]
Следовательно, $\mathbb Z[\sqrt5]$ не является факториальным кольцом. $\hfill \square$

% --- Problem 4 ---
\begin{task}
    \begin{enumerate}
	\item Чему равно 17-е натуральное число, дающие остатки 2, 4, 6, 8 от деления на 5, 9, 11, 14?

	\item Сколько решений в кольце $\mathbb{Z} / 360 \mathbb{Z}$ имеет уравнение $x^2 = 2$?

	\item Сколько решений в кольце $\mathbb{Z}/n\mathbb{Z}$ имеет уравнение $x^2 = 1$?
    \end{enumerate}
\end{task}

\sol
\soli{1}
Решим систему сравнений:
\[
x\equiv 2\pmod 5,\quad x\equiv 4\pmod 9,\quad x\equiv 6\pmod {11},\quad x\equiv 8\pmod {14}.
\]

Решим систему последовательной подстановкой:
\[
\begin{cases}
x\equiv 2\pmod 5,\\
x\equiv 4\pmod 9,\\
x\equiv 6\pmod{11},\\
x\equiv 8\pmod{14}.
\end{cases}
\]

Из первого сравнения:
\[
x=5k+2.
\]

Подставляя во второе,
\[
5k+2\equiv 4\pmod 9
\quad\Longrightarrow\quad
5k\equiv 2\pmod 9.
\]
Так как \(5^{-1}\equiv 2\pmod 9\), то
\[
k\equiv 4\pmod 9.
\]
Следовательно,
\[
x=5(9m+4)+2=45m+22.
\]

Теперь используем третье сравнение:
\[
45m+22\equiv 6\pmod{11}.
\]
Поскольку \(45\equiv 1\) и \(22\equiv 0\pmod{11}\), получаем
\[
m\equiv 6\pmod{11}.
\]
Значит,
\[
x=45(11n+6)+22=495n+292.
\]

Последнее условие:
\[
495n+292\equiv 8\pmod{14}.
\]
Так как \(495\equiv 5\) и \(292\equiv 12\pmod{14}\), имеем
\[
5n+12\equiv 8\pmod{14}
\quad\Longrightarrow\quad
5n\equiv 10\pmod{14}.
\]
Так как \(5^{-1}\equiv 3\pmod{14}\), то
\[
n\equiv 30\equiv 2\pmod{14}.
\]

Следовательно,
\[
x=495(14t+2)+292=6930t+1282.
\]

Натуральные числа (положительные) получаются при $t\ge 0$. Первое ($t=0$) равно $1282$, второе при $t=1$, и т.д.  
Следовательно, 17-е число соответствует $t=16$:
\[
x_{17}=1282+6930\cdot 16=1282+110880=112162.
\]

Ответ: $\boxed{112162}$
