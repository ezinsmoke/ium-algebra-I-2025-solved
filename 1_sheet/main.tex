% --- Problem 1 ---
\begin{task}
    \begin{enumerate}
    	\item Докажите, что множества $\mathbb{Z}$ и $2\mathbb{Z}$ изоморфны как абелевы группы относительно сложения. Изоморфны ли они как кольца?
	\item Изоморфны ли как абелевы группы $(\mathbb{R}, +)$ и $(\mathbb{Q}, +)$? $(\mathbb{Q}, +)$ и $(\mathbb{Z}, +)$?
    \end{enumerate}
\end{task}

\sol
\soli{1} Рассмотрим отображение
\[
\varphi: \mathbb{Z} \to 2\mathbb{Z}, \qquad \varphi(n) = 2n.
\]

Оно является:
\begin{enumerate}
    \item \textbf{Гомоморфизмом:} 
    \[
    \varphi(a+b)=2(a+b)=2a+2b=\varphi(a)+\varphi(b).
    \]
    \item \textbf{Инъекцией:} 
    \[
    \varphi(a)=\varphi(b) \implies 2a=2b \implies a=b.
    \]
    \item \textbf{Сюръекцией:} любой элемент $2k \in 2\mathbb{Z}$ имеет прообраз $k \in \mathbb{Z}$, так как $\varphi(k)=2k$.
\end{enumerate}

Следовательно, $\varphi$ — изоморфизм групп. Значит,
\[
\mathbb{Z} \cong 2\mathbb{Z}.
\]

\hfill $\square$

$\mathbb{Z}$ и 2$\mathbb{Z}$ неизоморфны как кольца. Предположим противное: пусть существует изоморфизм колец
\[
\varphi: \mathbb{Z} \to 2\mathbb{Z}.
\]

В любом кольце с единицей изоморфизм переводит единицу в единицу. Кольцо $\mathbb{Z}$ имеет единицу $1$. Значит,
\[
\varphi(1) = 1_{2\mathbb{Z}},
\]
где $1_{2\mathbb{Z}}$ — единичный элемент кольца $2\mathbb{Z}$.

Но в кольце $2\mathbb{Z}$ \emph{нет} единичного элемента: для любого $2k \in 2\mathbb{Z}$ выполнено
\[
(2k) \cdot 2 = 4k \neq 2k
\]
при $k \neq 0$, а при $k=0$ получаем $0\cdot 2 = 0$, но это не единица для других элементов.

Следовательно, $\varphi(1)$ не может существовать в $2\mathbb{Z}$. Противоречие.

Значит, изоморфизма колец не существует:
\[
\mathbb{Z} \not\cong 2\mathbb{Z}.
\]

\hfill $\square$

\newpage

\soli{2} 
\begin{enumerate}
    \item Используем \textbf{кардинальность}:
    \[
    |\mathbb{Q}| = \aleph_0, \qquad |\mathbb{R}| = 2^{\aleph_0}.
    \]
    Любой изоморфизм групп является биекцией, а биекция сохраняет мощность. Так как мощности различны, изоморфизма не существует:
    \[
    (\mathbb{Q}, +) \not\cong (\mathbb{R}, +).
    \]

    \hfill $\square$

    \item В группе $(\mathbb{Z}, +)$ все ненулевые элементы имеют бесконечный порядок, но группа порождается одним элементом ($\mathbb{Z} = \langle 1 \rangle$), то есть является циклической.

    Группа $(\mathbb{Q}, +)$ \emph{не является циклической}: предположим, что $\mathbb{Q} = \langle a/b \rangle$. Тогда все элементы имели бы вид $n \cdot (a/b)$, то есть были бы целыми кратными числа $a/b$. Но, например, число $a/(2b)$ не представимо в таком виде. Противоречие.

    Следовательно, они не изоморфны. \hfill $\square$
\end{enumerate}



% --- Problem 2 ---
\begin{task}
    \begin{enumerate}
	\item Докажите, что любая подгруппа $(\mathbb{Z}, +)$ имеет вид $n \mathbb{Z}$, где $n \geq 0$
	\item Найдите обратный элемент к $[157]$ в кольце $\mathbb{Z}/235\mathbb{Z}$ и все целые решения уравнения $157x + 235y = 11$. 
    \end{enumerate}
\end{task}


\sol
\soli{1}
Пусть $H \leqslant \mathbb{Z}$.

\begin{enumerate}
    \item Если $H = \{0\}$, то $H = 0\mathbb{Z}$.
    
    \item Пусть $H \neq \{0\}$. Тогда в $H$ есть ненулевые элементы. Так как $H$ — подгруппа, вместе с каждым элементом она содержит и его противоположный. Следовательно, в $H$ есть положительные элементы.
    
    Обозначим через $n$ \emph{наименьший положительный} элемент из $H$:
    \[
    n = \min\{ h \in H \mid h > 0 \}.
    \]
    
    Докажем, что $H = n\mathbb{Z}$.
    
    \begin{enumerate}
	\item \textbf{$n\mathbb{Z} \subseteq H$:} Так как $n \in H$, то $n + n + \dots + n \in H$ и $(-n) + (-n) + \dots \in H$. Значит, все кратные $n$ принадлежат $H$.
    
	\item \textbf{$H \subseteq n\mathbb{Z}$:} Возьмём произвольный $h \in H$. Разделим $h$ на $n$ с остатком:
	\[
	h = qn + r, \qquad 0 \leqslant r < n, \quad q \in \mathbb{Z}.
	\]
	Тогда
	\[
	r = h - qn.
	\]
	Здесь $h \in H$ и $qn \in H$ (так как $n \in H$), следовательно, $r \in H$.
    
	Но $0 \leqslant r < n$, а $n$ — наименьший положительный элемент в $H$. Значит, $r$ не может быть положительным. Следовательно, $r = 0$.
    
	Таким образом, $h = qn$, то есть $h \in n\mathbb{Z}$.
    \end{enumerate} 
    Получили $H = n\mathbb{Z}$, где $n \geqslant 0$. 
\end{enumerate}




Итак, любая подгруппа $\mathbb{Z}$ является циклической и порождается своим наименьшим положительным элементом. \hfill $\square$

\soli{2}
\[
\begin{aligned}
    235 &= 1 \cdot 157 + 78 \\
    157 &= 2 \cdot 78 + 1
\end{aligned}
\]

\[
\begin{aligned}
    1 &= 157 - 2 \cdot 78 \\
      &= 157 - 2 \cdot (235 - 157) \\
      &= 3 \cdot 157 - 2 \cdot 235
\end{aligned}
\]

Следовательно, \(157 \cdot 3 + 235 \cdot (-2) = 1\), значит \(n = 3\), \(k = -2\).

\[
    157 \cdot 3 \equiv 1 \pmod{235}
\]

\[
    \boxed{[157]^{-1} = [3]}
\]

\hfill $\square$

\[
\begin{aligned}
    157 \cdot 3 + 235 \cdot (-2) = 1 \, | \, \cdot 11\\
    157 \cdot 33 + 235 \cdot (-22) = 11
\end{aligned}
\]

значит, при $k \in \mathbb{N}$:
\[
    157 \cdot (k + 33) + 235 \cdot (k - 2) = 11
\]

тем самым $x = k + 33, y = k - 2$, при $k \in \mathbb{N}$

\hfill $\square$

% --- Problem 3 ---
\begin{task}
    Убедитесь, что $\mathbb{Z} [ \sqrt{5} ] = \{ a + b \sqrt{5} \, | \, a,b \in \mathbb{Z} \}$   является коммутативным кольцом с единицей без делителей нуля (областью целостности). Покажите, что это кольцо не является факториальным.
\end{task}

\sol
\soli{1}
Множество  
\[
   \{ \mathbb{Z} [ \sqrt{5} ] = \{ a + b \sqrt{5} \, | \,  a,b \in \mathbb{Z} \}
\]
замкнуто относительно сложения, вычитания и умножения:
\[
(a+b\sqrt5)(c+d\sqrt5)=(ac+5bd)+(ad+bc)\sqrt5,
\]
содержит $1=1+0\sqrt5$ и является подкольцом поля $\mathbb R$. Значит, это коммутативное кольцо с единицей и без делителей нуля, то есть область целостности. $\hfill \square$

\soli{2}
Рассмотрим норму
\[
N(a+b\sqrt5)=a^2-5b^2.
\]
Она мультипликативна: $N(\alpha\beta)=N(\alpha)N(\beta)$. Элемент является обратимым тогда и только тогда, когда его норма равна $\pm1$.

В кольце нет элементов нормы $\pm2$, так как сравнение
\[
a^2-5b^2\equiv a^2\pmod 5
\]
не может давать $2$ или $3$ по модулю $5$, а квадраты по модулю $5$ равны $0,1,4$.

Следовательно, любой элемент нормы $\pm4$ неприводим: если $\alpha=\beta\gamma$, то
\[
N(\beta)N(\gamma)=\pm4,
\]
но норм $\pm2$ нет, поэтому один из сомножителей имеет норму $\pm1$, то есть является единицей.

Имеем:
\[
N(2)=4,\qquad N(1+\sqrt5)=-4,\qquad N(\sqrt5-1)=-4,
\]
значит, $2$, $1+\sqrt5$, $\sqrt5-1$ неприводимы.

Но
\[
4=2\cdot 2=(1+\sqrt5)(\sqrt5-1).
\]
Проверим, что $2$ не ассоциирован с $1+\sqrt5$: если бы
\[
2=(c+d\sqrt5)(1+\sqrt5),
\]
то
\[
c+5d=2,\qquad c+d=0,
\]
откуда $d=\frac12\notin\mathbb Z$, противоречие. Аналогично $2$ не ассоциирован с $\sqrt5-1$.

Таким образом, число $4$ имеет два различных разложения на неприводимые множители:
\[
4=2\cdot 2=(1+\sqrt5)(\sqrt5-1).
\]
Следовательно, $\mathbb Z[\sqrt5]$ не является факториальным кольцом. $\hfill \square$

% --- Problem 4 ---
\begin{task}
    \begin{enumerate}
	\item Чему равно 17-е натуральное число, дающие остатки 2, 4, 6, 8 от деления на 5, 9, 11, 14?

	\item Сколько решений в кольце $\mathbb{Z} / 360 \mathbb{Z}$ имеет уравнение $x^2 = 2$?

	\item Сколько решений в кольце $\mathbb{Z}/n\mathbb{Z}$ имеет уравнение $x^2 = 1$?
    \end{enumerate}
\end{task}

\sol
\soli{1}
Решим систему сравнений:
\[
x\equiv 2\pmod 5,\quad x\equiv 4\pmod 9,\quad x\equiv 6\pmod {11},\quad x\equiv 8\pmod {14}.
\]

Решим систему последовательной подстановкой:
\[
\begin{cases}
x\equiv 2\pmod 5,\\
x\equiv 4\pmod 9,\\
x\equiv 6\pmod{11},\\
x\equiv 8\pmod{14}.
\end{cases}
\]

Из первого сравнения:
\[
x=5k+2.
\]

Подставляя во второе,
\[
5k+2\equiv 4\pmod 9
\quad\Longrightarrow\quad
5k\equiv 2\pmod 9.
\]
Так как \(5^{-1}\equiv 2\pmod 9\), то
\[
k\equiv 4\pmod 9.
\]
Следовательно,
\[
x=5(9m+4)+2=45m+22.
\]

Теперь используем третье сравнение:
\[
45m+22\equiv 6\pmod{11}.
\]
Поскольку \(45\equiv 1\) и \(22\equiv 0\pmod{11}\), получаем
\[
m\equiv 6\pmod{11}.
\]
Значит,
\[
x=45(11n+6)+22=495n+292.
\]

Последнее условие:
\[
495n+292\equiv 8\pmod{14}.
\]
Так как \(495\equiv 5\) и \(292\equiv 12\pmod{14}\), имеем
\[
5n+12\equiv 8\pmod{14}
\quad\Longrightarrow\quad
5n\equiv 10\pmod{14}.
\]
Так как \(5^{-1}\equiv 3\pmod{14}\), то
\[
n\equiv 30\equiv 2\pmod{14}.
\]

Следовательно,
\[
x=495(14t+2)+292=6930t+1282.
\]

Натуральные числа (положительные) получаются при $t\ge 0$. Первое ($t=0$) равно $1282$, второе при $t=1$, и т.д.  
Следовательно, 17-е число соответствует $t=16$:
\[
x_{17}=1282+6930\cdot 16=1282+110880=112162.
\]

Ответ: $\boxed{112162}$


\soli{2}
Так как $4 \mid 360$, из уравнения
\[
x^2 \equiv 2 \pmod{360}
\]
следует сравнение
\[
x^2 \equiv 2 \pmod{4}.
\]
Но квадраты целых чисел по модулю $4$ дают только остатки $0$ или $1$:
\[
(2k)^2 \equiv 0 \pmod{4}, \qquad (2k+1)^2 \equiv 1 \pmod{4}.
\]
Остаток $2$ невозможен. Следовательно, исходное уравнение не имеет решений. \hfill $\square$

\soli{3}
Выведем формулу для числа решений сравнения \(x^2\equiv1\pmod n\).

Пусть
\[
n=\prod_{i=1}^k p_i^{a_i}.
\]
По китайской теореме об остатках
\[
\mathbb Z/n\mathbb Z\cong
\prod_{i=1}^k\mathbb Z/p_i^{a_i}\mathbb Z.
\]
Поэтому число решений \(x^2\equiv1\pmod n\) равно произведению чисел решений сравнений
\[
x^2\equiv1\pmod{p_i^{a_i}}.
\]

Рассмотрим сначала нечётное простое \(p\). Имеем
\[
p^a\mid(x-1)(x+1).
\]
При этом
\[
\gcd(x-1,x+1)\mid2,
\]
поэтому \(p\nmid\gcd(x-1,x+1)\). Следовательно, вся степень \(p^a\) должна делить один из множителей:
\[
p^a\mid x-1
\quad\text{или}\quad
p^a\mid x+1.
\]
Таким образом,
\[
x\equiv\pm1\pmod{p^a},
\]
то есть имеется ровно \(2\) решения.

Теперь рассмотрим \(p=2\). При \(a=1\) имеется одно решение:
\[
x\equiv1\pmod2.
\]
При \(a=2\) имеются два решения:
\[
x\equiv\pm1\pmod4.
\]

Пусть \(a\ge3\). Из
\[
2^a\mid(x-1)(x+1)
\]
следует, что \(x\) нечётно. Числа \(x-1\) и \(x+1\) отличаются на \(2\), поэтому одно из них имеет \(2\)-адический порядок ровно \(1\), а другое должно иметь порядок не менее \(a-1\). Отсюда
\[
x\equiv\pm1\pmod{2^{a-1}}.
\]
Каждое из этих сравнений даёт два класса по модулю \(2^a\), поэтому всего получаем \(4\) решения:
\[
x\equiv\pm1,\quad
\pm(1+2^{a-1})\pmod{2^a}.
\]

Итак,
\[
f(p^a)=
\begin{cases}
1,&p=2,\ a=1,\\
2,&p=2,\ a=2,\\
4,&p=2,\ a\ge3,\\
2,&p\text{ нечётное}.
\end{cases}
\]

Следовательно,
\[
\boxed{
\#\{x\in\mathbb Z/n\mathbb Z:x^2=1\}
=
\prod_{p^a\parallel n}f(p^a)
}.
\]

% --- Problem 5 ---
\begin{task}
    Рассмотрим группу обратимых элементов $(\mathbb{Z}/n \mathbb{Z})^\times$. Определим функцию Эйлера формулой $\varphi(n) = | \mathbb{Z}/n \mathbb{Z}^\times |$.
    \begin{enumerate}
	\item Докажите \textit{теорему Эйлера}: $a^{\varphi(n)} = 1$ для любого $a \in (\mathbb{Z}/n \mathbb{Z})^\times$.

	\item Покажите, что $\varphi (n) = n (1 - p_1^{-1}) ... (1 - p_k^{-1})$, где $n = p_1^{\alpha_1} ... p_k^{\alpha_k}$, $p_i$ -- различные простые числа.
    \end{enumerate}
\end{task}

\sol
\soli{1}

Пусть 
\[
G=(\mathbb Z/n\mathbb Z)^\times
\]
— группа обратимых элементов кольца вычетов по модулю $n$. По определению,
\[
|G|=\varphi(n).
\]

Возьмём произвольный элемент $a\in G$. Рассмотрим отображение
\[
\mu_a\colon G\to G,\qquad x\mapsto ax.
\]
Оно является биекцией, поскольку умножение на обратимый элемент $a$ имеет обратное отображение $x\mapsto a^{-1}x$.

Выпишем все элементы группы $G$:
\[
G=\{g_1,g_2,\dots,g_m\},\qquad m=\varphi(n).
\]
Так как $\mu_a$ переставляет элементы $G$, то наборы
\[
g_1,g_2,\dots,g_m
\]
и
\[
a g_1,\ a g_2,\ \dots,\ a g_m
\]
совпадают с точностью до порядка. Перемножим все элементы:
\[
\prod_{i=1}^m g_i
=
\prod_{i=1}^m (a g_i)
=
a^m \prod_{i=1}^m g_i.
\]
Каждый элемент $g_i$ обратим, значит, их произведение $\prod_{i=1}^m g_i$ тоже обратимо. Сокращая на него, получаем
\[
1=a^m=a^{\varphi(n)}.
\]

Таким образом,
\[
\boxed{a^{\varphi(n)}=1 \quad \text{для любого } a\in (\mathbb Z/n\mathbb Z)^\times.}
\]

\soli{2}

Пусть
\[
n = p_1^{\alpha_1} p_2^{\alpha_2} \cdots p_k^{\alpha_k},
\]
где $p_1,\dots,p_k$ — различные простые числа, $\alpha_i \ge 1$.

Докажем, что
\[
\varphi(n) = n \prod_{i=1}^{k} \left(1 - \frac{1}{p_i}\right).
\]

Функция Эйлера мультипликативна: если $\gcd(a,b)=1$, то
\[
\varphi(ab) = \varphi(a)\varphi(b).
\]
Доказательство: по китайской теореме об остатках
\[
(\mathbb Z/ab\mathbb Z)^\times \cong (\mathbb Z/a\mathbb Z)^\times \times (\mathbb Z/b\mathbb Z)^\times,
\]
следовательно,
\[
\varphi(ab) = |(\mathbb Z/ab\mathbb Z)^\times| = |(\mathbb Z/a\mathbb Z)^\times| \cdot |(\mathbb Z/b\mathbb Z)^\times| = \varphi(a)\varphi(b).
\]


Для простого $p$ и $\alpha \ge 1$ имеем
\[
\varphi(p^\alpha) = p^\alpha - p^{\alpha-1} = p^\alpha \left(1 - \frac{1}{p}\right).
\]
Среди $1, ..., p^\alpha$ ровно $p^{\alpha - 1}$ делятся на $p$, а значит, ровно $p^\alpha - p^{\alpha-1}$ чисел взаимно просты с $p^\alpha$

Так как $p_i^{\alpha_i}$ попарно взаимно просты, используя мультипликативность, получаем
\[
\varphi(n) = \prod_{i=1}^{k} \varphi(p_i^{\alpha_i})
= \prod_{i=1}^{k} p_i^{\alpha_i} \left(1 - \frac{1}{p_i}\right)
= \left(\prod_{i=1}^{k} p_i^{\alpha_i}\right) \prod_{i=1}^{k} \left(1 - \frac{1}{p_i}\right)
= n \prod_{i=1}^{k} \left(1 - \frac{1}{p_i}\right).
\]

Таким образом,
\[
\boxed{\varphi(n) = n \prod_{p \mid n} \left(1 - \frac{1}{p}\right)}.
\]

% --- Problem 6 ---

\begin{task}
    Введём на множестве $X$ функций $f : \mathbb{Z}_{>0} \to \mathbb{R}$ две операции:
    \begin{itemize}
	\item поточечное сложение: $(f+g)(n) = f(n) + g(n)$;
	\item свёртка Дирихле: $(f * g)(n) = \sum_{d \mid n} f(d) g(\frac{n}{d})$.
    \end{itemize}
    \begin{enumerate}
	\item Докажите, что $(X, +, *)$ -- коммутативное кольцо с единицей (какая функция является единицей этого кольца?). Опишите все обратимые элементы

	\item Найдите обратный элемент функции 
	    \[1: \mathbb{Z}_{>0} \xrightarrow{n \mapsto 1} \mathbb{R} \]
	    Этот элемент называется \textit{функцией Мёбиуса}
    \end{enumerate}
\end{task}

\sol
\soli{1}

Рассмотрим множество
\[
X=\{f:\mathbb Z_{>0}\to \mathbb R\}.
\]
На нём заданы операции:
\[
(f+g)(n)=f(n)+g(n),\qquad 
(f*g)(n)=\sum_{d\mid n} f(d)g\!\left(\frac nd\right).
\]

Докажем, что $(X,+,*)$ -- коммутативное кольцо с единицей.

\paragraph{1. Аддитивная группа}
Сложение поточечное, поэтому $(X,+)$ -- абелева группа:
\begin{itemize}
\item сложение ассоциативно и коммутативно;
\item нулевой элемент -- функция $0(n)=0$;
\item противоположный к $f$ -- функция $-f$, $(-f)(n)=-f(n)$.
\end{itemize}

\paragraph{2. Свойства умножения (свёртки Дирихле)} проверим все свойства:

\textbf{Коммутативность:}
\[
(f*g)(n)=\sum_{d\mid n} f(d)g\!\left(\frac nd\right)
=\sum_{e\mid n} g(e)f\!\left(\frac ne\right)=(g*f)(n),
\]
где $e=n/d$.

\textbf{Ассоциативность:}
\[
((f*g)*h)(n)
=\sum_{d\mid n}(f*g)(d)h\!\left(\frac nd\right)
=\sum_{d\mid n}\sum_{a\mid d} f(a)g\!\left(\frac da\right)h\!\left(\frac nd\right).
\]
Полагая $d=ab$, $n=abc$, получаем
\[
((f*g)*h)(n)=\sum_{abc=n} f(a)g(b)h(c).
\]
Аналогично
\[
(f*(g*h))(n)=\sum_{abc=n} f(a)g(b)h(c),
\]
значит свёртка ассоциативна.

\textbf{Дистрибутивность:}
\[
(f*(g+h))(n)
=\sum_{d\mid n} f(d)(g+h)\!\left(\frac nd\right)
=\sum_{d\mid n} f(d)g\!\left(\frac nd\right)
+\sum_{d\mid n} f(d)h\!\left(\frac nd\right)
=(f*g+f*h)(n).
\]
Аналогично из коммутативности получаем $(f+g)*h=f*h+g*h$.

\paragraph{3. Единица кольца}

Единичным элементом является функция
\[
\varepsilon(n)=
\begin{cases}
1,& n=1,\\
0,& n>1.
\end{cases}
\]
Действительно,
\[
(\varepsilon*f)(n)=\sum_{d\mid n}\varepsilon(d)f\!\left(\frac nd\right)=f(n),
\]
так как единственное ненулевое слагаемое получается при $d=1$. Аналогично $f*\varepsilon=f$.

Следовательно, $(X,+,*)$ -- коммутативное кольцо с единицей $\varepsilon$.

\paragraph{4. Обратимые элементы}

Докажем, что
\[
f \text{ обратим } \iff f(1)\ne 0.
\]

\textbf{Необходимость.}
Пусть $f*g=\varepsilon$. Тогда при $n=1$:
\[
(f*g)(1)=f(1)g(1)=\varepsilon(1)=1,
\]
значит $f(1)\ne 0$.

\textbf{Достаточность.}
Пусть $f(1)\ne 0$. Построим обратную функцию $g$ рекурсивно.

Положим
\[
g(1)=\frac1{f(1)}.
\]
Для $n>1$ определим
\[
g(n)=-\frac1{f(1)}
\sum_{\substack{d\mid n\\ d>1}} f(d)\,g\!\left(\frac nd\right).
\]
Так как при $d>1$ число $n/d<n$, то $g(n/d)$ уже определено по индукции, значит $g$ корректно задана на всех $n$.

Тогда для $n=1$:
\[
(f*g)(1)=f(1)g(1)=1=\varepsilon(1).
\]
Для $n>1$:
\[
(f*g)(n)=\sum_{d\mid n} f(d)g\!\left(\frac nd\right)
=f(1)g(n)+\sum_{\substack{d\mid n\\ d>1}} f(d)g\!\left(\frac nd\right)=0=\varepsilon(n).
\]
Таким образом, $f*g=\varepsilon$, то есть $g$ -- обратный элемент к $f$.

Следовательно, обратимые элементы кольца $(X,+,*)$ -- это в точности функции $f:\mathbb Z_{>0}\to\mathbb R$ такие, что
\[
f(1)\ne 0.
\]  \hfill $\square$
